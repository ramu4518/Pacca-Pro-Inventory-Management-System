function increaseWeight(productId) { let weightSpan = document.getElementById(`weight_${productId}`); let currentWeight = parseFloat(weightSpan.textContent) || 0; let newWeight = currentWeight + 10; // Round to the nearest multiple of 10 weightSpan.textContent = Math.round(newWeight / 10) * 10; } function decreaseWeight(productId) { let weightSpan = document.getElementById(`weight_${productId}`); let currentWeight = parseFloat(weightSpan.textContent) || 0; if (currentWeight > 0) { let newWeight = currentWeight - 10; // Round to the nearest multiple of 10 weightSpan.textContent = Math.round(newWeight / 10) * 10; } }
